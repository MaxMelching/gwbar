    signalupshift=-1.2em,
    signalheight=1.5em,
    bottompaddingwithsubtitle=0.2em,
    bottompaddingnosubtitle=0.4em,
]{../gwbar}
\setbeamertemplate{frametitle}[gwbar]

% -- To see the impact of parameters like signalupshift or signalheight, it is
% -- very useful to change to styles of boxes used for filling since this is our

\usetikzlibrary{patterns}
\tikzset{
    gwbar@fillinglayer/.style={
        red,
		pattern=north west lines,
		pattern color=red,
	},
}
