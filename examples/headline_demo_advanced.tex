\def\gwheadlineshift{-1em}  % Represents shift we want to pad by
\def\gwheadlineshiftshift{0pt}

\addtobeamertemplate{frametitle}{}{%
%
\ifx\insertframesubtitle\@empty%
    % -- First if is important in case where signalupshift is non-zero
    % -- (and if no framesubtitle is set).
    \ifdim\gwbar@signalupshift=0pt
    \else
        \begin{tikzpicture}[
            remember picture,
            overlay,
            shift={($(current page.north west)+(0pt,\gwheadlineshiftshift)$)},
        ]
            \usebeamercolor[bg]{background}  % V1 to get correct color
                \draw[
                    % white,  % V2 to get correct color
                    fill,
                    % gwbar@fillinglayer,  % V3 to get correct color (and filling etc)
                    shift={(0, \gwbar@signalupshift)},
                ] (0, 0) rectangle (\paperwidth, \gwheadlineshift);
        \end{tikzpicture}
  \fi
\fi
% 
% -- Order matters for appearance! We do not want to cover GW signal
\gwbar%
% 
}
